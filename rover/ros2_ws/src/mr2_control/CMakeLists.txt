cmake_minimum_required(VERSION 3.16)
project(mr2_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(transmission_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(mr2_can_bus_core REQUIRED)
find_package(mr2_action_interface REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(mr2_control_node src/mr2_control_node.cpp)
target_include_directories(mr2_control_node PRIVATE include)
ament_target_dependencies(mr2_control_node rclcpp hardware_interface transmission_interface
  pluginlib mr2_can_bus_core mr2_action_interface geometry_msgs nav_msgs sensor_msgs
  std_msgs std_srvs diagnostic_msgs)
target_link_libraries(mr2_control_node yaml-cpp Threads::Threads)
install(TARGETS mr2_control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md BEHAVIOR.md VALIDATION.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_launch test/test_launch.py)
  find_package(controller_interface REQUIRED)
  find_package(mr2_devices_output_actuator REQUIRED)
  ament_add_gtest(test_drivers test/test_drivers.cpp)
  ament_target_dependencies(test_drivers rclcpp hardware_interface pluginlib mr2_can_bus_core sensor_msgs mr2_devices_output_actuator)
  option(MR2_ENABLE_TSAN "Instrument driver race test with ThreadSanitizer" OFF)
  if(MR2_ENABLE_TSAN)
    target_compile_options(test_drivers PRIVATE -fsanitize=thread -g)
    target_link_options(test_drivers PRIVATE -fsanitize=thread)
  endif()
  ament_add_gtest(test_converter test/test_converter.cpp)
  target_include_directories(test_converter PRIVATE include)
  ament_target_dependencies(test_converter rclcpp hardware_interface transmission_interface pluginlib mr2_can_bus_core sensor_msgs)
  ament_add_gtest(test_legacy_drive test/test_legacy_drive.cpp
    ../mr2_rover_control/src/twist_to_commands_controller.cpp)
  target_include_directories(test_legacy_drive PRIVATE include ../mr2_rover_control/include)
  ament_target_dependencies(test_legacy_drive controller_interface rclcpp geometry_msgs nav_msgs mr2_action_interface pluginlib)

  ament_add_gtest(test_processors test/test_processors.cpp)
  target_include_directories(test_processors PRIVATE include)
  target_link_libraries(test_processors Threads::Threads)
endif()
ament_package()
