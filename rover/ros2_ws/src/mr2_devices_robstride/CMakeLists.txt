cmake_minimum_required(VERSION 3.10)
project(mr2_devices_robstride LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(mr2_can_bus_core REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(robstride_device SHARED src/robstride_device.cpp)
target_include_directories(robstride_device PRIVATE include)
ament_target_dependencies(robstride_device mr2_can_bus_core hardware_interface pluginlib rclcpp)
pluginlib_export_plugin_description_file(mr2_can_bus_core plugin.xml)
install(TARGETS robstride_device LIBRARY DESTINATION lib)
add_executable(robstride_smoke_test src/robstride_smoke_test.cpp)
target_include_directories(robstride_smoke_test PRIVATE include)
ament_target_dependencies(robstride_smoke_test mr2_can_bus_core hardware_interface pluginlib rclcpp)
install(TARGETS robstride_smoke_test RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES plugin.xml README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_robstride test/test_robstride.cpp)
  target_include_directories(test_robstride PRIVATE include)
  ament_target_dependencies(test_robstride mr2_can_bus_core hardware_interface pluginlib rclcpp)
  ament_add_gtest(test_plugin_loading test/test_plugin_loading.cpp)
  ament_target_dependencies(test_plugin_loading mr2_can_bus_core hardware_interface pluginlib rclcpp)
endif()
ament_package()
